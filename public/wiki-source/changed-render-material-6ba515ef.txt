{{LSL Constant
|name=CHANGED_RENDER_MATERIAL
|type=integer
|value={{LSL Hex|0x1000}}
|desc=Prim [[PRIM_RENDER_MATERIAL|material ID]] or [[GLTF_Overrides|material overrides]] have changed on one or more faces. 

|examples=
<source lang="lsl2">
default
{
    changed(integer change)
    {
        if (change & CHANGED_RENDER_MATERIAL) //note that it's & and not &&... it's bitwise!
        {
            llOwnerSay("The prim's material ID and/or overrides have changed on one or more faces.");
        }
    }
}
</source>

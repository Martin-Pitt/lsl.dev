This script is the unofficial companion to the famous Hello Avatar script from Linden Lab. 
default
{
    state_entry()
    {
        llListen(0,"","","");
    }
    listen(integer chan, string name, key id, string msg)
    {
        if (msg == "Hello, Avatar!")
        {
        llSay(0, "Hello, " + name + "!");
        }
    }

}
